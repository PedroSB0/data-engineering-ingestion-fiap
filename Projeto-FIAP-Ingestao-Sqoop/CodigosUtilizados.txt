# --- ETAPA 1: EXPLORAÇÃO NA ORIGEM (MYSQL) ---
# Nota: Os comandos abaixo são executados dentro do console MySQL.
# mysql -u root -p retail_db
# Comandos executados: SHOW TABLES; DESCRIBE products; SELECT * FROM products LIMIT 10;

# --- ETAPA 2: VALIDAÇÃO DE CONECTIVIDADE ---
# Listando as tabelas do MySQL via Sqoop para garantir que a rede/acesso está OK
sqoop list-tables --connect jdbc:mysql://quickstart:3306/retail_db --username root --password cloudera

# --- ETAPA 3: PREPARAÇÃO DO AMBIENTE NO HDFS ---
# Criando o diretório de destino na camada Transient com permissões totais
sudo -u hdfs hadoop fs -mkdir -p /user/datalake/transient/retail
sudo -u hdfs hadoop fs -chmod 777 /user/datalake/transient/retail

# --- ETAPA 4: INGESTÃO DOS DADOS (IMPORT) ---
# Primeira tentativa de importação (sem o parâmetro de delete)
sqoop import --connect jdbc:mysql://quickstart:3306/retail_db --table products --warehouse-dir /user/datalake/transient/retail --username root --password cloudera

# --- ETAPA 5: VALIDAÇÃO DA CARGA ---
# Listando os arquivos gerados (part-m-XXXXX)
hdfs dfs -ls /user/datalake/transient/retail/products

# Lendo o final do primeiro arquivo para conferir os registros brutos
hadoop fs -tail /user/datalake/transient/retail/products/part-m-00000

# Contagem total de linhas para garantir que os 1345 registros foram importados
hadoop fs -cat /user/datalake/transient/retail/products/part-m-* | wc -l

# --- ETAPA 6: TRATAMENTO DE ERRO E RE-EXECUÇÃO ---
# Comando otimizado com --delete-target-dir para permitir re-execuções sem erro de "FileAlreadyExists"
sqoop import \
  --connect jdbc:mysql://quickstart:3306/retail_db \
  --table products \
  --warehouse-dir /user/datalake/transient/retail \
  --delete-target-dir \
  --username root \
  --password cloudera
